// lptrfRound runs one session round against a local openai-compatible server on one injected
// clock. answer is told the request number among the non-summarization requests; it advances
// the clock and returns the status and body of its answer. The Sleep seam does not wait.
func lptrfRound(t *testing.T, answer func(clock *modelMSClock, n int, summary bool) (int, string)) []api.Event {
	t.Helper()
	clock := &modelMSClock{t: time.Date(2026, 11, 20, 10, 0, 0, 0, time.UTC)}
	var n atomic.Int32
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		body, _ := io.ReadAll(r.Body)
		summary := strings.Contains(string(body), "You condense the earlier work")
		i := 0
		if !summary {
			i = int(n.Add(1))
		}
		status, text := answer(clock, i, summary)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(status)
		io.WriteString(w, text)
	}))
	defer srv.Close()
	client := provider.NewClient("test")
	client.Sleep = func(ctx context.Context, _ time.Duration) error { return ctx.Err() }
	root := t.TempDir()
	var events []api.Event
	s, err := NewSession(SessionConfig{
		Root: root, Role: provider.Role{Provider: "openai-compatible", Model: "local", BaseURL: srv.URL},
		Session: sys.NewWith([]string{"HOME=" + t.TempDir(), "PATH=" + os.Getenv("PATH")}, root),
		Client:  client,
		Emit:    func(e api.Event) { events = append(events, e) },
	})
	if err != nil {
		t.Fatal(err)
	}
	s.tools.Now = clock.now
	s.tools.Enabled = append(s.tools.Enabled, tools.ProbeTool)
	s.tools.Probe = func(string, any) (string, error) { return strings.Repeat("a", 20000), nil }
	res, err := s.Round(context.Background(), "go")
	if err != nil || res.Ending != EndTerminal {
		t.Fatalf("%+v %v", res, err)
	}
	return events
}

func lptrfCalls(names ...string) string {
	var parts []string
	for i, n := range names {
		args := `{}`
		if n == "finish" {
			args = `{\"summary\":\"done\"}`
		} else {
			args = `{\"command\":\"x\"}`
		}
		parts = append(parts, fmt.Sprintf(`{"id":"c%d","type":"function","function":{"name":"%s","arguments":"%s"}}`, i, n, args))
	}
	return fmt.Sprintf(`{"choices":[{"message":{"role":"assistant","content":null,"tool_calls":[%s]},"finish_reason":"tool_calls"}]}`, strings.Join(parts, ","))
}

// TestLPTRFAcceptance runs the scenario named by LPTRF_SCENARIO and prints what every event carried.
func TestLPTRFAcceptance(t *testing.T) {
	var events []api.Event
	switch os.Getenv("LPTRF_SCENARIO") {
	case "retry":
		events = lptrfRound(t, func(clock *modelMSClock, n int, _ bool) (int, string) {
			if n == 1 {
				clock.advance(100)
				return http.StatusInternalServerError, `{"error":{"message":"temporary"}}`
			}
			clock.advance(200)
			return http.StatusOK, lptrfCalls("finish")
		})
	case "condense":
		calls := make([]string, 20)
		for i := range calls {
			calls[i] = "probe"
		}
		events = lptrfRound(t, func(clock *modelMSClock, n int, summary bool) (int, string) {
			clock.advance(10)
			if summary {
				return http.StatusOK, `{"choices":[{"message":{"role":"assistant","content":"summary of earlier work"},"finish_reason":"stop"}]}`
			}
			if n <= 2 {
				return http.StatusOK, lptrfCalls(calls...)
			}
			return http.StatusOK, lptrfCalls("finish")
		})
	default:
		t.Fatal("LPTRF_SCENARIO not set")
	}
	type ev struct {
		Type   string `json:"type"`
		Has    bool   `json:"has"`
		GoType string `json:"go_type"`
		Value  string `json:"value"`
		JSON   string `json:"json"`
	}
	var out []ev
	for _, e := range events {
		v, has := e.Fields["model_ms"]
		j, _ := json.Marshal(v)
		out = append(out, ev{e.Type, has, fmt.Sprintf("%T", v), fmt.Sprint(v), string(j)})
	}
	data, _ := json.Marshal(out)
	fmt.Println("LPTRFOBS " + string(data))
}
