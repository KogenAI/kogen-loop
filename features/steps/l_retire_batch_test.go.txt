package loop

import (
 "context"
 "encoding/json"
 "fmt"
 "os"
 "testing"
 "time"
 "github.com/KogenAI/kogen-loop/sys"
 "github.com/KogenAI/kogen-loop/tools"
)

func TestAcceptRetireBatch(t *testing.T) {
 root := t.TempDir()
 tl := &tools.Tools{Root:root, Session:sys.NewWith(os.Environ(),root), Enabled:[]string{tools.ShellTool}}
 defer tl.EndShellRound()
 st := newStage(context.Background(),tl,nil)
 var handles []string
 for i:=0;i<3;i++ {
  marker:=fmt.Sprintf("ready%d",i)
  tl.ShellAfter=yieldAfter(t,root,marker)
  st.Done(i,olShell(fmt.Sprintf("c%d",i),"trap '' TERM; : > "+marker+"; while :; do :; done"))
  st.mu.Lock()
  g:=st.gen
  for len(g.slots)<=i || g.slots[i].state!=finished { st.cond.Wait() }
  text:=g.slots[i].out.Text
  st.mu.Unlock()
  var result struct { Status string `json:"status"`; Handle string `json:"handle"` }
  if err:=json.Unmarshal([]byte(text),&result); err!=nil || result.Status!="running" || result.Handle=="" { t.Fatalf("yield %d: %q %v",i,text,err) }
  handles=append(handles,result.Handle)
 }
 started:=time.Now()
 if os.Getenv("RETIRE_BATCH_CONTROL")=="round" { tl.EndShellRound() } else { st.Abort() }
 elapsed:=time.Since(started)
 // Check custody even when the latency assertion is red.
 for _,h:=range handles {
  out,err:=tl.Run(tools.ShellTool,map[string]any{"handle":h})
  if err!=nil || out.Text!="error: unknown shell handle" { t.Errorf("discarded handle %s: %q %v",h,out.Text,err) }
 }
 if groups:=tl.Session.LiveGroups(); len(groups)!=0 { t.Errorf("groups survived retire: %v",groups) }
 if elapsed>=10*time.Second { t.Errorf("retire exceeded two 5s grace periods: %v",elapsed) }
}
