// Driver tests of the Chat Completions response-header wait. They use a real local HTTP server,
// the package's simulated Client.Now and the signalTransport of idle_stream_test.go: the clock
// advances only while the client is blocked on the server, and nothing here sleeps.

const zzhdGuard = 5 * time.Second

// zzhdPlan is how the server answers one request: it holds the response headers until hold is
// closed (nil: sends them at once), then writes body's chunks until it is closed, or full.
type zzhdPlan struct {
	ctype string
	hold  chan struct{}
	body  chan string
	full  string
}

type zzhdServer struct {
	srv    *httptest.Server
	plans  []*zzhdPlan
	opened []chan struct{} // the request arrived
	gone   []chan struct{} // the handler returned: the answer ended or the client left
	mu     sync.Mutex
	n      int
	bodies []string
}

func newZZHDServer(t *testing.T, plans ...*zzhdPlan) *zzhdServer {
	t.Helper()
	s := &zzhdServer{plans: plans}
	for range plans {
		s.opened = append(s.opened, make(chan struct{}))
		s.gone = append(s.gone, make(chan struct{}))
	}
	s.srv = httptest.NewServer(http.HandlerFunc(s.serve))
	t.Cleanup(s.srv.Close)
	return s
}

func (s *zzhdServer) count() int {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.n
}

func (s *zzhdServer) serve(w http.ResponseWriter, r *http.Request) {
	b, _ := io.ReadAll(r.Body)
	s.mu.Lock()
	i := s.n
	s.n++
	s.bodies = append(s.bodies, string(b))
	s.mu.Unlock()
	if i >= len(s.plans) {
		http.Error(w, "unexpected extra request", 500)
		return
	}
	p := s.plans[i]
	defer close(s.gone[i])
	close(s.opened[i])
	if p.hold != nil {
		select {
		case <-p.hold:
		case <-r.Context().Done():
			return
		case <-time.After(2 * zzhdGuard):
			return
		}
	}
	w.Header().Set("Content-Type", p.ctype)
	w.WriteHeader(200)
	w.(http.Flusher).Flush()
	if p.body == nil {
		io.WriteString(w, p.full)
		return
	}
	for {
		select {
		case chunk, ok := <-p.body:
			if !ok {
				return
			}
			io.WriteString(w, chunk)
			w.(http.Flusher).Flush()
		case <-r.Context().Done():
			return
		case <-time.After(2 * zzhdGuard):
			return
		}
	}
}

// zzhdClient is one Respond in flight against the server.
type zzhdClient struct {
	t        *testing.T
	e        *env
	obs      *observer
	reads    chan readSignal
	guard    <-chan time.Time
	finished chan struct{}
	resp     *Response
	err      error
}

func zzhdStart(t *testing.T, srv *zzhdServer, mod func(*Request)) *zzhdClient {
	t.Helper()
	old := idlePoll
	idlePoll = 5 * time.Millisecond
	t.Cleanup(func() { idlePoll = old })
	e, _, req := obsEnv(t, "opencode-go")
	st := &signalTransport{rt: &http.Transport{}, reads: make(chan readSignal, 1024)}
	t.Cleanup(func() { st.rt.(*http.Transport).CloseIdleConnections() })
	e.c.HTTP = &http.Client{Transport: st}
	e.s = sys.NewWith([]string{"HOME=" + e.home, "PATH=" + os.Getenv("PATH"), "KOGEN_PROVIDER_URL=" + srv.srv.URL}, e.home)
	c := &zzhdClient{t: t, e: e, obs: &observer{}, reads: st.reads, guard: time.After(zzhdGuard), finished: make(chan struct{})}
	if mod != nil {
		mod(&req)
	}
	go c.respond(req)
	return c
}

func (c *zzhdClient) respond(req Request) {
	defer close(c.finished)
	c.resp, c.err = c.e.c.Respond(context.Background(), c.e.s, req, c.obs)
}

// again runs a second Respond on the same client after the first one returned.
func (c *zzhdClient) again(req Request) {
	c.t.Helper()
	c.finished = make(chan struct{})
	go c.respond(req)
}

func (c *zzhdClient) waitOpen(srv *zzhdServer, i int) {
	c.t.Helper()
	select {
	case <-srv.opened[i]:
	case <-c.guard:
		c.t.Fatalf("request %d never reached the server", i+1)
	}
}

// waitRead returns once the client has handled the first bytes of the attempt's body.
func (c *zzhdClient) waitRead(attempt, bytes int) {
	c.t.Helper()
	for {
		select {
		case sig := <-c.reads:
			if sig.attempt == attempt && sig.bytes >= bytes {
				return
			}
		case <-c.guard:
			c.t.Fatalf("attempt %d: the client never handled %d bytes of the answer", attempt, bytes)
		}
	}
}

func (c *zzhdClient) wait() {
	c.t.Helper()
	select {
	case <-c.finished:
	case <-c.guard:
		c.t.Fatal("Respond did not return")
	}
}

func zzhdChunk(delta map[string]any, finish any, extra map[string]any) string {
	choice := map[string]any{"index": int64(0), "delta": delta}
	if finish != nil {
		choice["finish_reason"] = finish
	}
	ch := map[string]any{"id": "chatcmpl-stream", "object": "chat.completion.chunk", "model": "kimi-k3", "choices": []any{choice}}
	for k, v := range extra {
		ch[k] = v
	}
	return "data: " + string(mustEncode(ch)) + "\n\n"
}

const zzhdDone = "data: [DONE]\n\n"

func zzhdUsage() map[string]any {
	return map[string]any{"usage": map[string]any{"prompt_tokens": int64(100), "completion_tokens": int64(20)}}
}

func zzhdTextAnswer(text string) (first, rest string) {
	return zzhdChunk(map[string]any{"role": "assistant", "content": text}, nil, nil),
		zzhdChunk(map[string]any{}, "stop", zzhdUsage()) + zzhdDone
}

func zzhdMessageText(t *testing.T, items []map[string]any) string {
	t.Helper()
	if len(items) == 0 || items[0]["type"] != "message" {
		t.Fatalf("items = %s, want an assistant message first", jsonOf(t, items))
	}
	list, _ := asList(items[0]["content"])
	if len(list) != 1 {
		t.Fatalf("message content = %v", items[0]["content"])
	}
	b, _ := asMap(list[0])
	s, _ := b["text"].(string)
	return s
}

func zzhdWantAnswer(t *testing.T, c *zzhdClient, srv *zzhdServer, text string) {
	t.Helper()
	if c.err != nil || c.resp == nil {
		t.Fatalf("the call failed: %v", idleText(c.err))
	}
	if got := zzhdMessageText(t, c.resp.Items); got != text {
		t.Fatalf("text = %q, want %q", got, text)
	}
	if n := srv.count(); n != 1 {
		t.Fatalf("%d requests reached the server, want 1 (an attempt was cut and retried)", n)
	}
	if len(c.e.clk.slept()) != 0 {
		t.Fatalf("slept %v: an attempt was retried", c.e.clk.slept())
	}
}

// Serial: idlePoll is a package global.
// The headers arrive after 60 s of simulated silence, then a normal streamed answer.
func TestHeaderDriverDelayed60s(t *testing.T) {
	first, rest := zzhdTextAnswer("late but whole")
	p := &zzhdPlan{ctype: "text/event-stream", hold: make(chan struct{}), body: make(chan string, 4)}
	srv := newZZHDServer(t, p)
	c := zzhdStart(t, srv, nil)
	c.waitOpen(srv, 0)
	c.e.clk.advance(60 * time.Second)
	close(p.hold)
	c.waitRead(1, 0)
	p.body <- first
	p.body <- rest
	close(p.body)
	c.wait()
	zzhdWantAnswer(t, c, srv, "late but whole")
}

// Serial: idlePoll is a package global.
// No headers for far longer than any header timeout: the error names the header wait.
func TestHeaderDriverTimeoutNamesHeaders(t *testing.T) {
	srv := newZZHDServer(t,
		&zzhdPlan{ctype: "text/event-stream", hold: make(chan struct{})},
		&zzhdPlan{ctype: "text/event-stream", hold: make(chan struct{})},
		&zzhdPlan{ctype: "text/event-stream", hold: make(chan struct{})})
	c := zzhdStart(t, srv, nil)
attempts:
	for i := range srv.plans {
		select {
		case <-srv.opened[i]:
		case <-c.finished:
			break attempts
		case <-c.guard:
			t.Fatalf("request %d never reached the server", i+1)
		}
		c.e.clk.advance(300 * time.Second)
		select {
		case <-srv.gone[i]:
		case <-c.guard:
			t.Fatalf("request %d: 300 s without headers and the client still waited", i+1)
		}
	}
	c.wait()
	if c.err == nil {
		t.Fatal("the call succeeded although the server never sent headers")
	}
	text := strings.ToLower(idleText(c.err))
	if !strings.Contains(text, "header") {
		t.Errorf("error %q does not say the wait was for response headers", text)
	}
	if strings.Contains(text, "idle") {
		t.Errorf("error %q uses the stream idle wording", text)
	}
}

// Serial: idlePoll is a package global.
// Headers after 100 s, the first chunk 150 s after them: 250 s in all, within the idle limit
// counted from the headers.
func TestHeaderDriverIdleClockStartsAtHeaders(t *testing.T) {
	first, rest := zzhdTextAnswer("slow first chunk")
	p := &zzhdPlan{ctype: "text/event-stream", hold: make(chan struct{}), body: make(chan string, 4)}
	srv := newZZHDServer(t, p)
	c := zzhdStart(t, srv, nil)
	c.waitOpen(srv, 0)
	c.e.clk.advance(100 * time.Second)
	close(p.hold)
	c.waitRead(1, 0)
	c.e.clk.advance(150 * time.Second)
	p.body <- first
	c.waitRead(1, len(first))
	p.body <- rest
	close(p.body)
	c.wait()
	zzhdWantAnswer(t, c, srv, "slow first chunk")
}

// A stream: true request answered with a plain JSON body (Content-Type application/json).
func TestHeaderDriverPlainJSONAnswer(t *testing.T) {
	body := string(mustEncode(goText("plain answer")))
	srv := newZZHDServer(t, &zzhdPlan{ctype: "application/json", full: body})
	c := zzhdStart(t, srv, nil)
	c.wait()
	zzhdWantAnswer(t, c, srv, "plain answer")
	if !strings.Contains(srv.bodies[0], `"stream":true`) {
		t.Fatalf("request body %q does not ask for a stream", srv.bodies[0])
	}
}

// A streamed tool call, then a second request that carries the tool result.
func TestHeaderDriverToolCallRoundTrip(t *testing.T) {
	const args = `{"path":"a.go","n":1}`
	frag := func(s string, first bool) string {
		call := map[string]any{"index": int64(0), "function": map[string]any{"arguments": s}}
		if first {
			call["id"], call["type"] = "call_rt", "function"
			call["function"].(map[string]any)["name"] = "read_file"
		}
		return zzhdChunk(map[string]any{"tool_calls": []any{call}}, nil, nil)
	}
	first := zzhdChunk(map[string]any{"role": "assistant"}, nil, nil) + frag(`{"path"`, true) + frag(`:"a.go"`, false) +
		frag(`,"n":`, false) + frag(`1}`, false) + zzhdChunk(map[string]any{}, "tool_calls", zzhdUsage()) + zzhdDone
	second, secondRest := zzhdTextAnswer("the file says hi")
	srv := newZZHDServer(t,
		&zzhdPlan{ctype: "text/event-stream", full: first},
		&zzhdPlan{ctype: "text/event-stream", full: second + secondRest})
	c := zzhdStart(t, srv, func(r *Request) { r.Tools = []any{goTool("read_file")} })
	c.wait()
	if c.err != nil || c.resp == nil {
		t.Fatalf("the first call failed: %v", idleText(c.err))
	}
	items := c.resp.Items
	last := items[len(items)-1]
	if last["type"] != "function_call" || last["call_id"] != "call_rt" || last["name"] != "read_file" || last["arguments"] != args {
		t.Fatalf("items = %s, want a function_call call_rt read_file %s", jsonOf(t, items), args)
	}
	input := user("x")
	for _, it := range items {
		input = append(input, it)
	}
	input = append(input, map[string]any{"type": "function_call_output", "call_id": "call_rt", "output": "hi from the tool"})
	c.again(Request{Role: goRole, Instructions: "p", Input: input, CacheID: "streamobs-cache", Tools: []any{goTool("read_file")}})
	c.wait()
	if c.err != nil || c.resp == nil {
		t.Fatalf("the second call failed: %v", idleText(c.err))
	}
	if got := zzhdMessageText(t, c.resp.Items); got != "the file says hi" {
		t.Fatalf("second answer = %q", got)
	}
	if srv.count() != 2 {
		t.Fatalf("%d requests, want 2", srv.count())
	}
	for _, want := range []string{`"tool_call_id":"call_rt"`, "hi from the tool", `"arguments":"{\"path\":\"a.go\",\"n\":1}"`} {
		if !strings.Contains(srv.bodies[1], want) {
			t.Errorf("second request body lacks %s: %s", want, srv.bodies[1])
		}
	}
}
