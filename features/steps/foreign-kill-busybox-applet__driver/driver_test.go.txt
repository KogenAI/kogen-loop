package sys

import (
	"encoding/json"
	"fmt"
	"os"
	"testing"
)

func TestForeignKillBusyboxDriver(t *testing.T) {
	command := os.Getenv("FKB_COMMAND")
	rule, refused := ForeignKill(command, func(pid int) bool { return pid == 4242 })
	b, err := json.Marshal(map[string]any{"command": command, "rule": rule, "refused": refused})
	if err != nil {
		t.Fatal(err)
	}
	fmt.Println("FKB " + string(b))
}
